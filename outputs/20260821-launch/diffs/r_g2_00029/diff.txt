#!/bin/bash
# Pre-commit hook to run mypy on all Python files in src/
# Blocks the commit if type-checking fails

set -e

echo "Running mypy on src/ Python files..."

# Run mypy on all Python files in src/
/anaconda/envs/azureml_py310_sdkv2/bin/mypy src/

if [ $? -eq 0 ]; then
    echo "Type checking passed. Proceeding with commit..."
else
    echo "Type checking failed. Aborting commit."
    exit 1
fi
