#!/bin/bash
# Pre-commit hook to run mypy on Python files in src/

# Find all Python files in src/
python_files=$(find src/ -name "*.py" -type f)

# Run mypy on all Python files
mypy "$python_files"

# Check if mypy returned non-zero (indicating errors)
if [ $? -ne 0 ]; then
    echo "Type checking failed. Please fix mypy errors before committing."
    exit 1
fi

echo "Type checking passed."
exit 0
