

--- .git/hooks/pre-commit ---
#!/bin/bash

# Pre-commit hook to run mypy on all Python files in src/
# Blocks commit if type-checking fails

echo "Running mypy type checking on src/..."

# Run mypy on all Python files in src/
python -m mypy src/

# Check if mypy failed (non-zero exit code)
if [ $? -ne 0 ]; then
    echo "ERROR: mypy type checking failed. Please fix the errors before committing."
    exit 1
fi

echo "Type checking passed!"
