#!/bin/bash
# Pre-commit hook to run mypy on all Python files in src/

# Get list of Python files in src/
python_files=$(git ls-files src/ -- '*.py' 2>/dev/null | grep -v py.typed)

if [ -z "$python_files" ]; then
    echo "No Python files in src/ to check"
    exit 0
fi

# Run mypy on the Python files
echo "Running mypy on src/ Python files..."
python -m mypy "$python_files"

# Check mypy exit code
if [ $? -ne 0 ]; then
    echo "mypy check failed, aborting commit"
    exit 1
fi

echo "mypy check passed"
exit 0
