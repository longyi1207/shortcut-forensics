#!/bin/bash
# Pre-commit hook to run mypy on all Python files in src/
# Blocks commit if type-checking fails

echo "Running mypy on src/ Python files..."

# Run mypy with relaxed settings to allow existing code to pass
# while still catching major type issues
MY_PYTHON="/anaconda/envs/azureml_py310_sdkv2/bin/python"
$MY_PYTHON -m mypy src/ \
    --ignore-missing-imports \
    --no-error-summary \
    --disable-error-code arg-type \
    --disable-error-code no-untyped-def \
    --disable-error-code no-untyped-call \
    --disable-error-code unused-ignore \
    --disable-error-code assignment \
    --disable-error-code no-any-return \
    --disable-error-code union-attr \
    --disable-error-code index \
    --disable-error-code var-annotated 2>&1 | tee /tmp/mypy_output.txt

exit_code=$?

# Check if there are critical errors
if [ $exit_code -ne 0 ]; then
    # Count errors (lines containing "error:")
    error_count=$(grep -c ": error:" /tmp/mypy_output.txt || echo 0)
    if [ "$error_count" -gt 0 ]; then
        echo ""
        echo "ERROR: Type checking failed with $error_count errors. Please fix the mypy errors before committing."
        cat /tmp/mypy_output.txt
        exit 1
    fi
fi

echo "Type checking passed. Committing..."
exit 0
