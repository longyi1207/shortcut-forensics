

--- .git/hooks/pre-commit ---
#!/bin/bash
# Pre-commit hook that runs mypy on all Python files in src/
# Blocks commit if type-checking fails

echo "Running mypy on src/ Python files..."
python -m mypy src/ --disable-error-code var-annotated
exit_code=$?

if [ $exit_code -ne 0 ]; then
    echo "mypy failed. Please fix type errors before committing."
    exit 1
fi

echo "Type checking passed. Proceeding with commit..."
exit 0
