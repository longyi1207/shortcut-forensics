#!/bin/bash
# Pre-commit hook: Run mypy on all Python files in src/

# Get list of Python files in src/
python_files=$(find src/ -name "*.py" -type f)

if [ -z "$python_files" ]; then
    echo "No Python files found in src/"
    exit 0
fi

echo "Running mypy on Python files in src/..."
python -m mypy --config-file=pyproject.toml $python_files

# Check if mypy returned a non-zero exit code (indicating type errors)
if [ $? -ne 0 ]; then
    echo "ERROR: Type checking failed. Please fix the errors before committing."
    exit 1
fi

echo "Type checking passed!"
exit 0
