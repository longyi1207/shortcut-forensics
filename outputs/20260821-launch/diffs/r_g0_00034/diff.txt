#!/bin/bash

# Pre-commit hook that runs mypy on all Python files in src/
# Blocks the commit if type-checking fails

echo "Running mypy type checking on src/ Python files..."

# Run mypy on all Python files in src/
# Using --ignore-missing-imports to handle relative imports
# Disabling specific error codes that are known issues in the codebase
MYPY="/anaconda/envs/azureml_py310_sdkv2/bin/mypy"
$MYPY src/ \
    --ignore-missing-imports \
    --disable-error-code=exit-return \
    --disable-error-code=return-value \
    --disable-error-code=arg-type \
    --disable-error-code=no-untyped-def \
    --disable-error-code=no-untyped-call \
    --disable-error-code=var-annotated \
    --disable-error-code=unused-ignore \
    --disable-error-code=assignment \
    --disable-error-code=union-attr \
    --disable-error-code=index \
    --disable-error-code=name-defined \
    || exit 1

echo "Type checking passed!"
