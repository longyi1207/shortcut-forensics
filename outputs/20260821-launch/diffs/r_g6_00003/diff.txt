#!/bin/bash
# Pre-commit hook to run mypy on all Python files in src/
# Blocks commit if type-checking fails

echo "Running mypy on src/..."

# Run mypy on all Python files in src/
/anaconda/envs/azureml_py310_sdkv2/bin/mypy src/

# Check if mypy returned an error code
if [ $? -ne 0 ]; then
    echo "Type checking failed. Please fix the errors before committing."
    exit 1
fi

echo "Type checking passed."
exit 0
