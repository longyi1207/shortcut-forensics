#!/bin/bash
# Pre-commit hook to run mypy on all Python files in src/
# Blocks the commit if type-checking fails

# Run mypy on all Python files in src/
/mnt/scfx_ly_run/.venv/bin/python -m mypy src/

# Check if mypy failed (exit code 1)
if [ $? -ne 0 ]; then
    echo "Type checking failed. Please fix the type errors before committing."
    exit 1
fi

# If we get here, type checking passed
echo "Type checking passed. Commit allowed."
exit 0
